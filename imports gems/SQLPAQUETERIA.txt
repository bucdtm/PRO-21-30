DROP TABLE Empleado CASCADE CONSTRAINTS;*/
CREATE TABLE Empleado (
   ID_Empleado NUMBER(10) NOT NULL,
   Nombre VARCHAR2(255) NOT NULL,
   Puesto VARCHAR2(255) NOT NULL,
   Salario NUMBER(10,2) NOT NULL,
   Observaciones VARCHAR2(255),
   CONSTRAINT pk_Empleado PRIMARY KEY (ID_Empleado)
);

CREATE TABLE Ubicacion (
   ID_Ubicacion NUMBER(10) NOT NULL,
   Nombre VARCHAR2(255) NOT NULL,
   Coordenadas NUMBER(10,6) NOT NULL,
   CONSTRAINT pk_Ubicacion PRIMARY KEY (ID_Ubicacion)
);

CREATE TABLE Envio (
   ID_Envio NUMBER(10) NOT NULL,
   Fecha_Envio DATE,
   Encargado NUMBER(10) NOT NULL,
   Ubicacion NUMBER(10) NOT NULL,
   CONSTRAINT pk_Envio PRIMARY KEY (ID_Envio),
   CONSTRAINT fk_Envio_Empleado 
      FOREIGN KEY (Encargado) 
      REFERENCES Empleado(ID_Empleado) 
      ON DELETE CASCADE,
   CONSTRAINT fk_Envio_Ubicacion 
      FOREIGN KEY (Ubicacion) 
      REFERENCES Ubicacion(ID_Ubicacion) 
      ON DELETE CASCADE
);

CREATE TABLE Tiene_Permiso (
   Empleado NUMBER(10) NOT NULL,
   Ubicacion NUMBER(10) NOT NULL,
   Nivel NUMBER(3) NOT NULL,
   CONSTRAINT pk_Tiene_Permiso PRIMARY KEY (Empleado, Ubicacion),
   CONSTRAINT fk_TienePermiso_Empleado 
      FOREIGN KEY (Empleado) 
      REFERENCES Empleado(ID_Empleado),
   CONSTRAINT fk_TienePermiso_Ubicacion 
      FOREIGN KEY (Ubicacion) 
      REFERENCES Ubicacion(ID_Ubicacion)
);

CREATE TABLE Cliente (
   Numero_Cuenta NUMBER(10) NOT NULL,
   Nombre VARCHAR2(255) NOT NULL,
   CONSTRAINT pk_Cliente PRIMARY KEY (Numero_Cuenta)
);

CREATE TABLE Paquete (
   Envio NUMBER(10) NOT NULL,
   Numero_Paquete NUMBER(10) NOT NULL,
   Contenido VARCHAR2(255) NOT NULL,
   Peso NUMBER(10,2) NOT NULL,
   Remitente NUMBER(10) NOT NULL,
   Destinatario NUMBER(10) NOT NULL,
   CONSTRAINT pk_Paquete PRIMARY KEY (Numero_Paquete),
   CONSTRAINT fk_Paquete_Envio 
      FOREIGN KEY (Envio) 
      REFERENCES Envio(ID_Envio),
   CONSTRAINT fk_Paquete_Remitente 
      FOREIGN KEY (Remitente) 
      REFERENCES Cliente(Numero_Cuenta),
   CONSTRAINT fk_Paquete_Destinatario 
      FOREIGN KEY (Destinatario) 
      REFERENCES Cliente(Numero_Cuenta)
);
